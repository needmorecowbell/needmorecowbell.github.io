{{- $baseURL := .Page.Site.Params.S3CDN -}} {{- $id := .Get "id" | default (printf "gallery-%d" (index (seq 1 1) 0)) -}} {{- $path := .Get "path" | default "" -}} {{- $thumbnailSize := .Get "thumbnailSize" | default "250" -}} {{- $images := .Get "images" | default "" -}} {{- $galleryTitle := .Get "title" | default "Photo gallery" -}} {{- /* Build the base URL from S3CDN and path */ -}} {{- $itemsBaseURL := $baseURL -}} {{- if $path -}} {{- $itemsBaseURL = printf "%s/%s" $baseURL (strings.TrimSuffix "/" $path) -}} {{- end -}} {{- /* Gallery grid styles are in assets/css/custom.css - use CSS variable for custom thumbnail size */ -}}
{{- if $images -}} {{- range $index, $imgName := split $images "," -}} {{- $img := strings.TrimSpace $imgName -}} {{- /* Check if this is a video file */ -}} {{- $isVideo := or (strings.HasSuffix $img ".mp4") (strings.HasSuffix $img ".MP4") (strings.HasSuffix $img ".mov") (strings.HasSuffix $img ".MOV") (strings.HasSuffix $img ".webm") (strings.HasSuffix $img ".avi") -}} {{- /* All media uses .thumb.jpg as thumbnail (images and videos) */ -}} {{- $ext := path.Ext $img -}} {{- $thumb := printf "%s.thumb.jpg" (strings.TrimSuffix $ext $img) -}} {{- $fullURL := printf "%s/%s" $itemsBaseURL $img -}} {{- $thumbURL := printf "%s/%s" $itemsBaseURL $thumb -}} {{- /* Generate accessible alt text */ -}} {{- $altBase := path.Base $img | strings.TrimSuffix $ext -}} {{- $altText := replace $altBase "_" " " | replace "-" " " -}} {{- if $isVideo -}} {{- else -}} {{- /* Check if this is a GIF - use original as thumbnail since GIFs are animated */ -}} {{- $isGif := or (strings.HasSuffix $img ".gif") (strings.HasSuffix $img ".GIF") -}} {{- $displayThumb := $thumbURL -}} {{- if $isGif -}} {{- $displayThumb = $fullURL -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- /* Support for inner content (legacy format) */ -}} {{ .Inner }} {{- end -}}
